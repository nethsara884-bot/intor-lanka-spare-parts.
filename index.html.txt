        // PERSISTENCE LOCAL STORAGE
        function loadProducts() {
            const stored = localStorage.getItem('intor_lanka_products');
            if (stored) {
                try {
                    products = JSON.parse(stored);
                } catch (e) {
                    products = DEFAULT_PRODUCTS;
                }
            } else {
                products = DEFAULT_PRODUCTS;
                saveProductsToStorage();
            }
        }

        function saveProductsToStorage() {
            localStorage.setItem('intor_lanka_products', JSON.stringify(products));
        }

            const container = document.getElementById('products-grid');
            const searchQuery = (document.getElementById('catalog-search').value || '').toLowerCase().trim();

            let filtered = products.filter(p => {
                const matchesCat = selectedCategory === 'All' || p.category === selectedCategory;
                const matchesQuery = !searchQuery || 
                    p.title.toLowerCase().includes(searchQuery) || 
                    p.sku.toLowerCase().includes(searchQuery) ||
                    p.brand.toLowerCase().includes(searchQuery);
                return matchesCat && matchesQuery;
            });

            if (filtered.length === 0) {
                container.innerHTML = `
                    <div class="col-span-full bg-white p-12 text-center rounded-xl border border-dashed border-slate-300">
                        <i class="fa-solid fa-box-open text-4xl text-slate-300 mb-3"></i>

        // PRODUCT MODAL DETAILS
        function openProductModal(id) {
            const p = products.find(item => item.id === id);
            if (!p) return;

            currentModalProduct = p;
            document.getElementById('modal-category').textContent = p.category;
            document.getElementById('modal-title').textContent = p.title;
            document.getElementById('modal-sku').textContent = p.sku;
            document.getElementById('modal-brand').textContent = p.brand;
            document.getElementById('modal-price').textContent = p.price;
            document.getElementById('modal-description').textContent = p.desc || 'Contact INTOR LANKA for detailed vehicle compatibility and technical specifications.';
            
            // Stock badge
            const stockBadge = document.getElementById('modal-stock-badge');
            stockBadge.textContent = p.stock;
            stockBadge.className = `absolute top-3 left-3 text-[10px] font-bold px-2.5 py-1 rounded-full uppercase ${p.stock === 'In Stock' ? 'bg-green-100 text-green-800' : 'bg-amber-100 text-amber-800'}`;

            // Photos
            document.getElementById('modal-main-img').src = p.img1;
            document.getElementById('modal-thumb-1').src = p.img1;
            document.getElementById('modal-thumb-2').src = p.img2;

            // Pre-formatted WhatsApp Order Link
            const waText = encodeURIComponent(`Hello INTOR LANKA, I would like to place an order / enquiry for:\nProduct: ${p.title}\nSKU: ${p.sku}\nBrand: ${p.brand}\nPrice: ${p.price}`);
            document.getElementById('modal-wa-btn').href = `https://wa.me/94778800709?text=${waText}`;

            document.getElementById('product-modal').classList.remove('hidden');
        }

            document.getElementById('product-form-container').classList.remove('hidden');
            const form = document.getElementById('product-form');
            form.reset();

            if (id) {
                const p = products.find(item => item.id === id);
                if (p) {
                    document.getElementById('form-title').textContent = "Edit Spare Part";
                    document.getElementById('form-product-id').value = p.id;
                    document.getElementById('form-title-input').value = p.title;
                    document.getElementById('form-sku-input').value = p.sku;
                    document.getElementById('form-category-input').value = p.category;
                    document.getElementById('form-brand-input').value = p.brand;
                    document.getElementById('form-price-input').value = p.price;
                    document.getElementById('form-stock-input').value = p.stock;
                    document.getElementById('form-img1-input').value = p.img1;
                    document.getElementById('form-img2-input').value = p.img2;
                    document.getElementById('form-desc-input').value = p.desc;

        function saveProduct(e) {
            e.preventDefault();
            const id = document.getElementById('product-form-id')?.value || document.getElementById('form-product-id').value;
            
            const newProd = {
                id: id || Date.now().toString(),
                title: document.getElementById('form-title-input').value,
                sku: document.getElementById('form-sku-input').value,
                category: document.getElementById('form-category-input').value,
                brand: document.getElementById('form-brand-input').value,
                price: document.getElementById('form-price-input').value,
                stock: document.getElementById('form-stock-input').value,
                img1: document.getElementById('form-img1-input').value,
                img2: document.getElementById('form-img2-input').value,
                desc: document.getElementById('form-desc-input').value
            };

            if (id) {
                const index = products.findIndex(p => p.id === id);
                if (index !== -1) products[index] = newProd;
            } else {
                products.unshift(newProd);
            }

            saveProductsToStorage();
            renderAdminTable();
            renderProducts();
            closeProductForm();
            alert('Product saved successfully to persistent inventory database!');
        }

        function editProduct(id) {
            openProductForm(id);
        }

        function deleteProduct(id) {
            if (confirm('Are you sure you want to delete this product?')) {
                products = products.filter(p => p.id !== id);
                saveProductsToStorage();
                renderAdminTable();
                renderProducts();
            }
        }
